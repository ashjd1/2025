PSQL

install the psql the run command  "$ psql" to enter the data base
"$ \l"                                     to list the all availabel database 
"$ \c <daatabase name>"                    to connect (change) the existing data base 
"$ create database test"                   to create new database with test name
"$ drop database company"                  to delete  database
"$ drop database test with (force);"       to delete no matter what
"$ \d <name>"                              to describe table or database
"$ \dt "                                   to list down the all tables from database.

auto increament data type is "serial", even if you dont mention the vlaue, it will automaticelly takes the next value. 

CREATE TABLE employees(
emp_id SERIAL PRIMARY KEY,                             PRIMARY KEY dont allow null vlaue
fname VARCHAR(50) NOT NULL,
lname VARCHAR(50) NOT NULL,
email VARCHAR(100) NOT NULL UNIQUE,                    must be UNIQUE evey time.
dept VARCHAR(50),
salary DECIMAL(10,2) DEFAULT 30000.00,
hire_date DATE NOT NULL DEFAULT CURRENT_DATE           if you dont provide the date then it will take the value as DEFAULT CURRENT_DATE
);

Stored procudure is like a function in sql, and you can all the function with name of that procedure

CREATE OR REPLACE PROCEDURE update_emp_salary(p_employee_id INT, p_new_salary NUMERIC)
LANGUAGE plpgsql
AS $$
BEGIN
UPDATE employees SET salary = p_new_salary WHERE emp_id = p_employee_id;
END;
$$;

update_emp_salary(3, 75000);

CREATE OR REPLACE PROCEDURE add_employee(
p_fname VARCHAR,
p_Iname VARCHAR,
p_email VARCHAR,
p_dept VARCHAR,
p_salary NUMERIC

LANGUAGE plpgsql

BEGIN
INSERT INTO employees (fname, lname, email, dept, salary)

END;
$$;

CREATE OR REPLACE FUNCTION dept_max_sal_emp1(dept_name VARCHAR)
RETURNS TABLE(emp_id INT, fname VARCHAR, salary NUMERIC)
AS $$
BEGIN
RETURN QUERY
SELECT
e.emp_id, e.fname, e.salary
FROM
employees e
WHERE
e.dept = dept_name
AND e.salary = (
SELECT MAX(emp.salary)
FROM employees emp
WHERE emp.dept = dept_name

);
END;
$$ LANGUAGE plpgsql;
